# 5. Clean column names
titanic <- janitor::clean_names(titanic)

# 6. Handle missing values
# Cabin has very high missingness, so it is removed.
titanic_clean <- titanic %>%
  select(-cabin) %>%
  mutate(
    age = ifelse(is.na(age), median(age, na.rm = TRUE), age),
    embarked = ifelse(is.na(embarked),
                      names(sort(table(embarked), decreasing = TRUE))[1],
                      embarked)
  )

# 7. Convert variables to appropriate data types
titanic_clean <- titanic_clean %>%
  mutate(
    survived = factor(survived, levels = c(0,1),
                      labels = c("No", "Yes")),
    pclass = factor(pclass, levels = c(1,2,3),
                    labels = c("1st", "2nd", "3rd")),
    sex = factor(sex),
    embarked = factor(embarked),
    family_size = sib_sp + parch + 1
  )

# 8. Outlier detection using IQR
iqr_outliers <- function(x) {
  q1 <- quantile(x, 0.25, na.rm = TRUE)
  q3 <- quantile(x, 0.75, na.rm = TRUE)
  iqr <- q3 - q1
  lower <- q1 - 1.5 * iqr
  upper <- q3 + 1.5 * iqr
  list(
    lower = lower,
    upper = upper,
    count = sum(x < lower | x > upper, na.rm = TRUE)
  )
}

age_outliers <- iqr_outliers(titanic_clean$age)
fare_outliers <- iqr_outliers(titanic_clean$fare)
age_outliers
fare_outliers

# We do not automatically delete all outliers because extreme fares can
# represent genuine passengers/tickets. The report documents them first.

# 9. Normalization (Min-Max scaling)
min_max <- function(x) {
  (x - min(x, na.rm = TRUE)) /
    (max(x, na.rm = TRUE) - min(x, na.rm = TRUE))
}

titanic_clean <- titanic_clean %>%
  mutate(
    age_scaled = min_max(age),
    fare_scaled = min_max(fare)
  )
